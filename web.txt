local HttpService = game:GetService("HttpService")
local MarketplaceService = game:GetService("MarketplaceService")

local request = (syn and syn.request) or (http and http.request) or http_request or (fluxus and fluxus.request) or request

local GITHUB_RAW_URL = "https://raw.githubusercontent.com/2btrleans/scr1ipt/main/url.txt"

local successUrl, tunnelUrl = pcall(function()
    return game:HttpGet(GITHUB_RAW_URL)
end)

if successUrl and tunnelUrl and tunnelUrl:find("trycloudflare.com") then
    tunnelUrl = tunnelUrl:gsub("%s+", "")

    local veriTablosu = {
        ["kullanici"] = game.Players.LocalPlayer.Name,
        ["place_id"] = game.PlaceId,
        ["oyun_adi"] = (pcall(function() return MarketplaceService:GetProductInfo(game.PlaceId).Name end) and MarketplaceService:GetProductInfo(game.PlaceId).Name or "Bilinmiyor")
    }

    local response = request({
        Url = tunnelUrl .. "/webhook",
        Method = "POST",
        Headers = {
            ["Content-Type"] = "application/json",
            ["User-Agent"] = "Mozilla/5.0 (Windows NT 10.0; Win64; x64) AppleWebKit/537.36 (KHTML, like Gecko) Chrome/122.0.0.0 Safari/537.36"
        },
        Body = HttpService:JSONEncode(veriTablosu)
    })

    if response and response.StatusCode == 200 then
    else
    end
else
end
